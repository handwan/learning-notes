    系统编程：文件、进程与信号¶

    POSIX/Linux 系统调用视角：文件描述符、出错处理、fork/exec、信号、线程（pthread）。代码都是可直接编译的 C。

    本篇只依赖 POSIX/Linux 接口，示例自成一体。

    1. 系统调用与 errno¶

    系统调用 = 用户程序请求内核做事（读文件、开进程、发网络包）。手册对应 man 2 xxx（系统调用）与 man 3 xxx（C 库函数）。

    返回值的惯例：失败返回 -1（或 NULL），并把错误码放进全局 errno。

    #include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>

int fd = open("/no/such/file", O_RDONLY);
if (fd == -1) {
    // errno 是线程局部的；要用就马上存下来（别的调用可能覆盖它）
    int err = errno;
    fprintf(stderr, "open failed: %s (errno=%d)\n", strerror(err), err);
    perror("open");        // 等价但更省事：open: No such file or directory
}

    规则：

    • 每个系统调用都要检查返回值（read 也可能失败，不只是"没读到"）
    • errno 只在失败时有意义，且会被后续调用覆盖——先存再用
    • 常见 errno：ENOENT（不存在）、EACCES（权限）、EINTR（被信号打断）、EAGAIN（非阻塞暂无数据）、EEXIST

    2. 文件描述符与文件 IO¶

    fd 是进程内"打开文件表"的索引（小整数）。约定俗成：0 stdin、1 stdout、2 stderr。

    #include <fcntl.h>
#include <unistd.h>
#include <string.h>
#include <stdio.h>

int main(void) {
    // open：flags 组合；权限只在 O_CREAT 时用（0644）
    int fd = open("demo.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    if (fd == -1) { perror("open"); return 1; }

    const char* msg = "hello fd\n";
    ssize_t n = write(fd, msg, strlen(msg));
    if (n == -1) { perror("write"); close(fd); return 1; }
    // n 是"实际写了多少字节"，可能小于请求（短写）——循环写才是完整写法

    // lseek：定位（SEEK_SET/CUR/END），返回新偏移
    off_t end = lseek(fd, 0, SEEK_END);
    printf("file size = %ld\n", (long)end);

    close(fd);
    return 0;
}

    要点：

    • O_APPEND 的原子性：多进程/多线程同时往一个文件追加日志时，用 O_APPEND 保证"定位到末尾 + 写"是原子的；自己 lseek + write 会有竞争
    • 短读/短写：read/write 返回值可能小于请求（管道、信号、磁盘），要循环到写完
    • EINTR：被信号打断时 read 可能返回 -1 + EINTR——循环重试（或用 SA_RESTART，见信号节）
    • open 的 flags 速记：O_RDONLY/O_WRONLY/O_RDWR + O_CREAT/O_TRUNC/O_APPEND/O_EXCL（配合 O_CREAT：文件已存在就失败，做"锁文件"）
    • 写管道/套接字的原子性：不超过 PIPE_BUF（Linux 4K）的写是原子的

    pread / pwrite：带偏移，不改共享文件位置¶

    ssize_t pread(int fd, void* buf, size_t n, off_t offset);
ssize_t pwrite(int fd, const void* buf, size_t n, off_t offset);

    pread/pwrite 不修改 fd 的当前偏移，多个线程共享同一个 fd 读不同区域时不会互相抢 offset；read/write 会改共享偏移。

    dup / dup2：重定向的底层¶

    #include <fcntl.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    int fd = open("out.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    dup2(fd, STDOUT_FILENO);      // 把 fd 复制到 1 号位置：之后 printf 都进文件
    close(fd);                    // 原来的 fd 可以关掉（1 号还开着）
    printf("this goes to out.txt\n");
    // shell 里的 `cmd > out.txt` 就是这么实现的：fork + dup2 + exec
    return 0;
}

    fcntl 常用：F_SETFD, FD_CLOEXEC（exec 时自动关掉这个 fd，防 fd 泄漏给子进程——现代推荐直接用 open(..., O_CLOEXEC)）；F_GETFL/F_SETFL 改 O_NONBLOCK。

    ulimit -n                 # 一个进程能开多少 fd（值因系统而异，先查；高并发服务要调大）
ls /proc/self/fd/         # 当前进程打开了哪些 fd（排查 fd 泄漏）

    3. 文件元数据与目录¶

    #include <sys/stat.h>
#include <stdio.h>

int main(void) {
    struct stat st;
    if (stat("demo.txt", &st) == -1) { perror("stat"); return 1; }

    printf("size = %ld\n", (long)st.st_size);
    printf("mode = %o\n", st.st_mode & 07777);          // 权限位
    printf("is regular file: %d\n", S_ISREG(st.st_mode));
    printf("is dir: %d\n", S_ISDIR(st.st_mode));
    printf("owner uid = %d, inode = %lu\n", st.st_uid, (unsigned long)st.st_ino);
    return 0;
}
    调用 区别
    stat 跟随符号链接（看目标）
    lstat 不跟随（看链接本身）
    fstat 用已打开的 fd 查（避免 TOCTOU：先开再查更安全）

    遍历目录（opendir/readdir）：

    #include <dirent.h>
#include <stdio.h>
#include <string.h>

int main(void) {
    DIR* d = opendir(".");
    if (!d) { perror("opendir"); return 1; }
    struct dirent* e;
    while ((e = readdir(d)) != NULL) {
        if (strcmp(e->d_name, ".") == 0 || strcmp(e->d_name, "..") == 0) continue;
        printf("%s\n", e->d_name);             // 以 . 开头的隐藏文件会正常列出
    }
    closedir(d);
    return 0;
}

    监听文件变化：inotify（配置热加载、日志采集常用）：

    #include <sys/inotify.h>
#include <unistd.h>
#include <stdio.h>

int main(void) {
    int fd = inotify_init1(IN_CLOEXEC);
    if (fd == -1) { perror("inotify_init1"); return 1; }

    int wd = inotify_add_watch(fd, "/tmp", IN_CREATE | IN_DELETE | IN_MODIFY);
    if (wd == -1) { perror("inotify_add_watch"); close(fd); return 1; }

    char buf[4096];
    ssize_t n = read(fd, buf, sizeof(buf));      // 阻塞直到有事件
    if (n == -1) { perror("read"); inotify_rm_watch(fd, wd); close(fd); return 1; }

    printf("got event (%ld bytes)\n", (long)n);
    inotify_rm_watch(fd, wd);
    close(fd);
    return 0;
}

    （更省事的方式：inotifywait 命令，来自 inotify-tools。）

    4. 进程：fork / exec / wait¶

    #include <stdio.h>
#include <stdlib.h>
#include <sys/wait.h>
#include <unistd.h>

int main(void) {
    pid_t pid = fork();                 // 调用一次，返回两次
    if (pid == -1) { perror("fork"); return 1; }

    if (pid == 0) {
        // 子进程：写时复制（COW）——fork 后内存是"共享只读，写时复制"
        execlp("echo", "echo", "hello from child", (char*)NULL);
        perror("execlp");               // exec 成功就不会走到这
        _exit(127);                     // 失败退出（_exit 不刷 stdio 缓冲，fork 后更安全）
    }

    // 父进程
    int status = 0;
    pid_t done = waitpid(pid, &status, 0);      // 收尸，防止僵尸
    if (done == -1) { perror("waitpid"); return 1; }
    if (WIFEXITED(status)) {
        printf("child exited with %d\n", WEXITSTATUS(status));
    } else if (WIFSIGNALED(status)) {
        printf("child killed by signal %d\n", WTERMSIG(status));
    }
    return 0;
}

    要点：

    • fork 返回两次：父进程得到子 PID，子进程得到 0
    • fork 后立刻 exec 是经典组合；fork 的坑在于复制整个地址空间（COW 缓解，但多线程程序里 fork 很危险——只允许调用异步信号安全的函数）
    • wait/waitpid 必须做，否则子进程退出后变僵尸（SIGCHLD 到来时回收）
    • exec 家族：execl/execlp/execv/execvp/execve——l 是参数列表，v 是数组，p 是查 PATH，e 是自定义环境
    • system() 会起 shell（有注入风险）；返回值是 wait status，不是直接退出码——用 WIFEXITED / WEXITSTATUS 解析；popen() 适合"读命令输出"
    • _exit vs exit：_exit 不跑 atexit/不刷 stdio 缓冲——fork 后的子进程里用 _exit 更安全（避免重复刷父进程的缓冲）

    5. 信号¶

    #include <signal.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>

volatile sig_atomic_t g_stop = 0;          // 信号处理器里只能碰这种类型

static void on_term(int sig) {
    (void)sig;
    g_stop = 1;                            // 只设标志，别在处理器里干活
}

int main(void) {
    struct sigaction sa;
    memset(&sa, 0, sizeof(sa));
    sa.sa_handler = on_term;
    sigemptyset(&sa.sa_mask);
    sa.sa_flags = SA_RESTART;              // 被信号打断的系统调用自动重启
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGTERM, &sa, NULL);

    while (!g_stop) {
        pause();                           // 等信号（被 SIGNAL 打断返回 -1 + EINTR）
    }
    printf("graceful shutdown\n");
    return 0;
}

    规则（很重要）：

    • 用 sigaction 而不是 signal（语义在各系统上不一致）
    • 处理器里只能调用异步信号安全的函数（write 可以，printf/malloc 不行）——通常只设一个 volatile sig_atomic_t 标志，主循环里检查
    • SA_RESTART 让被中断的 read/write 自动重启；不设就自己处理 EINTR
    • 多线程程序用 sigwait/signalfd 更安全：先 pthread_sigmask 屏蔽，再在专用线程里同步等待
    // 发信号
kill(pid, SIGTERM);            // 给进程
raise(SIGINT);                 // 给自己
alarm(5);                      // 5 秒后给自己 SIGALRM（简单超时）

    异步信号安全之外：SIGKILL(9)/SIGSTOP(19) 不可捕获；SIGSEGV 这类不要试图"恢复"（直接崩了更好排查）。

    6. 线程：pthread 最小集¶

    #include <pthread.h>
#include <stdint.h>
#include <stdio.h>

static void* worker(void* arg) {
    long id = (long)(intptr_t)arg;
    printf("thread %ld\n", id);
    return NULL;
}

int main(void) {
    pthread_t t[4];
    for (long i = 0; i < 4; ++i) pthread_create(&t[i], NULL, worker, (void*)(intptr_t)i);
    for (int i = 0; i < 4; ++i) pthread_join(t[i], NULL);   // 不 join 就 detach，别放着不管
    return 0;
}

    gcc main.c -pthread -o app        # 编译要加 -pthread

    要点：

    • 同步原语：pthread_mutex_t（互斥）、pthread_cond_t（条件变量）、pthread_rwlock_t（读写锁）——加锁-改-解锁，比信号量直觉
    • 和 fork 的坑：fork 后子进程只剩调用它的那个线程、锁状态可能损坏——多线程程序优先 posix_spawn
    • C++ 的 std::thread/std::mutex 底层就是这些

    7. 调试：strace 看系统调用¶

    strace ./a.out                          # 跟踪所有系统调用
strace -f ./a.out                       # 跟随子进程（fork/exec）
strace -e trace=openat,read,write ./a.out     # 只看文件相关
strace -e trace=network ./a.out         # 只看网络
strace -p <pid>                         # 挂到运行中的进程（排"卡住"问题特别有用）
strace -c ./a.out                       # 统计每个系统调用的次数/耗时

    常见用法：程序卡住 → strace -p 看它停在哪个调用；"文件找不到/权限不够" → -e trace=file 看它到底 open 了哪些路径（比猜快得多）。

    8. core dump 与 coredumpctl¶

    进程被信号杀死时（SIGSEGV / SIGABRT），内核可以把内存快照写成 core 文件，留作事后分析：

    ulimit -c unlimited                  # 允许生成（默认常常是 0，不生成）
cat /proc/sys/kernel/core_pattern    # core 去哪；systemd 系统会交给 systemd-coredump

coredumpctl list                     # 列出崩溃记录（时间、PID、信号、命令行）
coredumpctl info <pid>               # 概要：信号、命令行、栈顶几帧
coredumpctl gdb <pid>                # 直接用 gdb 打开现场

    "服务半夜挂了"的排查第一步：coredumpctl list 看有没有崩溃记录，再用 info / gdb 看死在哪。

    9. 要点回顾¶

    1. 每个系统调用都查返回值，errno 要立刻保存；perror/strerror 打印
    2. fd 是小整数（0/1/2 是标准流）；O_APPEND 保证追加原子性；读写要处理短读短写和 EINTR
    3. dup2 是重定向的底层（cmd > file = fork + dup2 + exec）；O_CLOEXEC 防 fd 泄漏给子进程
    4. 元数据用 stat/lstat/fstat（区别在是否跟随链接）；目录用 opendir/readdir
    5. inotify 监听文件变化（配置热加载、日志采集）
    6. fork 返回两次（父得子 PID、子得 0，COW 内存）；必须 waitpid 收尸否则僵尸
    7. exec 成功后不返回；子进程失败退出用 _exit
    8. 信号用 sigaction；处理器里只设 volatile sig_atomic_t 标志（只能用异步信号安全函数）；多线程用 sigwait/signalfd
    9. strace 是系统编程的 X 光：-p 看卡住、-e trace=file 看路径、-c 看统计
    10. 线程：pthread 四件（create/join/mutex/cond）；编译加 -pthread；fork 与线程混用要小心
    11. core dump：ulimit -c 放开、coredumpctl list/info/gdb 看崩溃现场
