    树与堆¶

    树把"线性查找"变成"分叉查找"（O(log n)）；堆则专门服务"每次拿最值"。本篇：BST → 平衡树 → 堆 → Trie。

    1. 树的基本词汇¶

            root(根)
        /      \
     child    child
      /  \      |
   leaf leaf  subtree(子树)
    词 含义
    深度/高度 根到节点的层数 / 到最远叶子的层数
    度 一个节点有几个孩子
    完全二叉树 除了最后一层都填满，且最后一层靠左（堆的数组表示靠它）
    满二叉树 每层都填满（节点数 2^h - 1）

    二叉树的数组表示（下标从 0）：i 的左孩子 2i+1、右孩子 2i+2、父亲 (i-1)/2——堆用它，省指针。

    2. 二叉搜索树（BST）：查找路径¶

    性质：左子树全部 < 根 < 右子树全部。

    • 查找/插入/删除：沿一条路径走 → O(h)（h 是树高）
    • h 取决于插入顺序——实测（10 万个数）：
    插入方式 树高
    有序插入（1,2,3,…） 100000（退化成链表，O(n)）
    随机插入 41（≈ 2·log₂n，O(log n)）

    这就是"平衡"要解决的问题：树高失控 → 查找退化。关键认知：复杂度是"树的形状"决定的，不是"BST"这个名字保证的。

    手写实现：

    struct BSTNode {
    int key; BSTNode *left, *right;
    explicit BSTNode(int k) : key(k), left(nullptr), right(nullptr) {}
};
struct BST {
    BSTNode* root = nullptr;
    void insert(int k) {                      // 迭代插入（递归在退化时会爆栈）
        BSTNode** c = &root;
        while (*c) c = (k < (*c)->key) ? &(*c)->left : &(*c)->right;
        *c = new BSTNode(k);
    }
    bool find(int k) const { BSTNode* c = root; while (c) { if (k == c->key) return true; c = k < c->key ? c->left : c->right; } return false; }
    void erase(int k) { root = erase(root, k); }
    BSTNode* erase(BSTNode* t, int k) {       // 教材三种情况
        if (!t) return nullptr;
        if (k < t->key) t->left = erase(t->left, k);
        else if (k > t->key) t->right = erase(t->right, k);
        else {
            if (!t->left)  { BSTNode* r = t->right; delete t; return r; }   // ① 无左孩子
            if (!t->right) { BSTNode* l = t->left;  delete t; return l; }   // ② 无右孩子
            BSTNode** s = &t->right; while ((*s)->left) s = &(*s)->left;    // ③ 中序后继
            t->key = (*s)->key; BSTNode* succ = *s; *s = succ->right; delete succ;
        }
        return t;
    }
    ~BST() { destroy(root); }
    static void destroy(BSTNode* t) { if (!t) return; destroy(t->left); destroy(t->right); delete t; }
};
// 实测：三种删除后中序仍有序；后继自己带右孩子的刁钻情况也不丢节点

    旋转到底怎么转（AVL 四种情况）：

    失衡分四种（以"左子树更高"为例）：
LL：左孩子的左边插入      → 一次右旋
LR：左孩子的右边插入      → 先对左孩子左旋，再对自己右旋
RR：右孩子的右边插入      → 一次左旋
RL：右孩子的左边插入      → 先对右孩子右旋，再对自己左旋

    struct AVLNode {
    int key, h; AVLNode *left, *right;
    explicit AVLNode(int k) : key(k), h(1), left(nullptr), right(nullptr) {}
};
struct AVL {
    AVLNode* root = nullptr;
    static int  H(AVLNode* t)   { return t ? t->h : 0; }                       // 空树高度 0
    static void fix(AVLNode* t) { int a = H(t->left), b = H(t->right); t->h = 1 + (a > b ? a : b); }
    static AVLNode* rot_right(AVLNode* y) {          // 右旋（LL 用）
        AVLNode* x = y->left; y->left = x->right; x->right = y; fix(y); fix(x); return x;
    }
    static AVLNode* rot_left(AVLNode* x) {           // 左旋（RR 用）
        AVLNode* y = x->right; x->right = y->left; y->left = x; fix(x); fix(y); return y;
    }
    void insert(int k) { root = insert(root, k); }
    AVLNode* insert(AVLNode* t, int k) {
        if (!t) return new AVLNode(k);
        if (k < t->key) t->left = insert(t->left, k);
        else if (k > t->key) t->right = insert(t->right, k);
        else return t;
        fix(t);                                      // 回溯时更新高度
        int bal = H(t->left) - H(t->right);
        if (bal > 1)  { if (k > t->left->key) t->left = rot_left(t->left);    return rot_right(t); }  // LL / LR
        if (bal < -1) { if (k < t->right->key) t->right = rot_right(t->right); return rot_left(t); }  // RR / RL
        return t;
    }
};
// 实测：四种插入序列旋转后根都是 20；10 万有序插入高度 ≤ 20（同样数据 BST 是 100000 层）

           9
      / \
     7   8       数组: [9,7,8,3,5,...]
    / \
   3   5
    操作 复杂度 说明
    看堆顶 O(1) 最大/最小值
    push（上浮） O(log n) 放到末尾一路和父亲比
    pop（下沉） O(log n) 堆顶和末尾交换，再把新顶沉下去
    建堆 heapify O(n) 从最后一个非叶子往前下沉（不是 O(n log n)！）

    实测：200 万个随机数取 Top-10（手写堆）：

    做法 耗时
    全部入堆再弹出（堆排序） 基准
    只维护"大小为 10 的小顶堆" 快约 190 倍

    Top-K 套路：求最大 K 个 → 用小顶堆（堆顶是 K 个里最小的，新元素比堆顶大才替换）；求最小 K 个反之。

    手写实现（小顶堆的 sift up/down）：

    /* 数组即完全二叉树 —— i 的孩子是 2i+1 / 2i+2 */
struct Heap {
    int* a; int n, cap;
    explicit Heap(int cap_) : a(new int[cap_]), n(0), cap(cap_) {}
    ~Heap() { delete[] a; }

    void push(int v) {                              // 上浮 O(log n)
        if (n == cap) {                             // 满了翻倍
            int nc = cap * 2; int* na = new int[nc];
            for (int i = 0; i < n; ++i) na[i] = a[i];
            delete[] a; a = na; cap = nc;
        }
        int i = n++; a[i] = v;
        while (i) { int p = (i - 1) / 2; if (a[p] <= a[i]) break;
                    int t = a[p]; a[p] = a[i]; a[i] = t; i = p; }
    }
    int pop() {                                     // 下沉 O(log n)
        int top = a[0]; a[0] = a[--n];
        int i = 0;
        for (;;) { int m = i, l = 2*i + 1, r = 2*i + 2;
            if (l < n && a[l] < a[m]) m = l;
            if (r < n && a[r] < a[m]) m = r;
            if (m == i) break; int t = a[i]; a[i] = a[m]; a[m] = t; i = m; }
        return top;
    }
};
// 实测：乱序 push 后连续 pop，输出严格有序（堆排序）

            root
       /    \
      c      d
      |      |
      a      o
      |      |
      t   →  dog
    → cat

    实测（20 万个随机 8 字母单词，插入+查找各 20 万次）：

    结构 总耗时
    Trie 慢
    开放寻址哈希（整个词做 key） 快几十倍

    前缀统计（"以 xx 开头的词有几个"）：Trie ✅ 独有 / 哈希 ❌ 做不到

    认知：Trie 在"纯查词"上不如哈希表（节点多、指针跳），它的价值在前缀族操作：

    • 自动补全、前缀计数、模糊匹配（配合 Levenshtein）
    • tokenizer（BPE 的字典查找）、IP 路由表（最长前缀匹配）

    优化：压缩成 Radix Tree/ Patricia Trie（把单链合并，省节点）、双数组 Trie（数组化，查得快）。

    手写实现：

    struct TrieNode {
    TrieNode* ch[26]; bool end;
    TrieNode() : end(false) { for (int i = 0; i < 26; ++i) ch[i] = nullptr; }
};
struct Trie {
    TrieNode* root = new TrieNode();
    void insert(const char* w) {
        TrieNode* p = root;
        for (; *w; ++w) { int i = *w - 'a'; if (!p->ch[i]) p->ch[i] = new TrieNode(); p = p->ch[i]; }
        p->end = true;                              // 词尾标记（区分"前缀"和"完整的词"）
    }
    bool find(const char* w) const {
        TrieNode* p = root;
        for (; *w; ++w) { p = p->ch[*w - 'a']; if (!p) return false; }
        return p->end;                              // 走到底还得是词尾才算"找到"
    }
};
// 实测：cat/car/dog 插入后，find(ca)=false（只是前缀）、find(cat)=true

    6. B 树 / B+ 树¶

    B 树家族 = "多路平衡查找树"：一个节点存几百个 key（而不是二叉树那样 2 个孩子），把树高压到 3~4 层——一次磁盘页访问恰好读一层，这正是数据库索引要的形状。

    m 阶 B 树的定义：

    约束 内容
    每个节点最多 m − 1 个 key、m 个孩子
    除根外每个节点最少 ⌈m/2⌉ − 1 个 key（根至少 1 个）
    所有叶子 同一层（绝对平衡）
    节点内 key 有序，孩子子树落在相邻 key 之间的区间

    插入（先插入、满了再分裂）：

    ① 一路找到叶子，插入 key（节点可能暂时有 m 个 key）
② 若节点溢出（> m-1）：取中间 key 升到父节点，左右各留 ⌈m/2⌉-1 个
③ 父节点也可能因此溢出 → 继续往上分裂；根分裂时树长高一层

    /* m 阶 B 树：每节点最多 m-1 个 key、m 个孩子（这里 m = 5） */
const int M = 5;
struct BNode {
    int n, key[M]; BNode* child[M + 1]{}; bool leaf;
    BNode(bool lf = true) : n(0), leaf(lf) {}
};
struct BTree {
    BNode* root = new BNode();

    static int lower(BNode* x, int k) { int i = 0; while (i < x->n && x->key[i] < k) ++i; return i; }
    bool search(BNode* x, int k) const {                      // 一路向下找
        int i = lower(x, k);
        if (i < x->n && x->key[i] == k) return true;
        return x->leaf ? false : search(x->child[i], k);
    }
    bool insert_rec(BNode* x, int k) {                        // 返回 true = x 溢出需父节点处理
        if (x->leaf) {                                        // 叶子里插入（可能暂时 m 个）
            int i = x->n - 1;
            while (i >= 0 && x->key[i] > k) { x->key[i + 1] = x->key[i]; --i; }
            x->key[i + 1] = k; ++x->n;
            return x->n > M - 1;
        }
        int i = lower(x, k);
        if (insert_rec(x->child[i], k)) {                      // 孩子分裂：中间 key 升上来
            BNode* y = x->child[i]; int mid = y->n / 2;
            BNode* z = new BNode(y->leaf); z->n = y->n - mid - 1;
            for (int j = 0; j < z->n; ++j) z->key[j] = y->key[mid + 1 + j];
            if (!y->leaf) for (int j = 0; j <= z->n; ++j) z->child[j] = y->child[mid + 1 + j];
            int up = y->key[mid]; y->n = mid;
            for (int j = x->n; j > i; --j) { x->key[j] = x->key[j - 1]; x->child[j + 1] = x->child[j]; }
            x->key[i] = up; x->child[i + 1] = z; ++x->n;
            return x->n > M - 1;
        }
        return false;
    }
    void insert(int k) {
        if (insert_rec(root, k)) {                            // 根溢出 → 树长高
            BNode* y = root; int mid = y->n / 2;
            BNode* z = new BNode(y->leaf); z->n = y->n - mid - 1;
            for (int j = 0; j < z->n; ++j) z->key[j] = y->key[mid + 1 + j];
            if (!y->leaf) for (int j = 0; j <= z->n; ++j) z->child[j] = y->child[mid + 1 + j];
            int up = y->key[mid]; y->n = mid;
            BNode* s = new BNode(false);
            s->key[0] = up; s->child[0] = y; s->child[1] = z; s->n = 1;
            root = s;
        }
    }
};
// 实测（5 阶、2 万随机 key）：树高 8，分裂 7396 次
// 校验：全部可查 + 中序有序 + 每个节点占用满足 (⌈m/2⌉-1, m-1)

    删除（了解规则即可，代码不展开）：先删，若节点 key 数低于下限 → 向兄弟借（父节点下来一个、兄弟上来一个）；兄弟也刚好在下限 → 和兄弟合并（父节点少一个 key，可能连锁向上）。

    B+ 树 vs B 树（数据库为什么用 B+）：

    B 树 B+ 树
    数据位置 每个节点都可能存数据 只有叶子存数据，内部只导航
    key 是否重复 不重复 内部 key 是叶子的副本（分裂时"复制"上去）
    范围查询 要走树回溯 叶子链表顺序扫
    查找稳定性 命中高度不固定 每次都要走到叶子（高度固定）
    典型用途 文件系统（如 ext4 的 htree） 数据库索引 
    /* B+ 树要点（在上面 B 树基础上改两处）：
   ① 分裂时，右节点的第一个 key 复制到父节点（不是从子节点搬走）
   ② 叶子用 next_leaf 串成链表，范围查询沿链表扫即可 */
struct BPNode { int n, key[M]; BPNode* child[M + 1]{}; BPNode* next_leaf = nullptr; bool leaf; ... };

    实测（5 阶、5 万随机 key）：沿叶子链表扫描得到 50000 条有序记录（这就是"范围查询 O(log n + 结果数)"的含义）。

    和数据库的连接：库里的索引就是这个东西的工程版——实测 100 万行主键索引只有 2~3 层；页 = 节点、分裂 = 页分裂，细节见存储引擎与日志篇。

    7. 结构选型¶

    需求 结构 备注
    有序遍历、范围、前驱后继 平衡树（AVL/红黑树） O(log n) 稳定
    每次只要最值 / Top-K 堆 小顶堆求最大 K 个
    前缀、自动补全、tokenizer Trie / Radix Tree 纯查词不如哈希
    纯查找、无需顺序 哈希表 平均 O(1)，无序（见后文哈希表篇）

    8. 要点回顾¶

    1. BST 的复杂度由树高决定：有序插入退化到 100000 层（实测），随机插入才 41 层
    2. 平衡树是"用旋转换稳定"：AVL 严、红黑树松（库实现多用它）；生产里用现成实现，不用手写
    3. 堆 = 完全二叉树 + 数组：push/pop O(log n)、建堆 O(n)；Top-K 用小顶堆（实测比全排序快两个数量级）
    4. Trie 的价值在前缀：纯查词不如哈希（慢几十倍），前缀统计/补全/tokenizer 才用它
    5. B+ 树解决"磁盘上的树"：高扇出压树高，叶子链表做范围扫描（数据库篇）
    6. 选型：有序 → 平衡树/跳表；只要最值 → 堆；前缀 → Trie；纯查找 → 哈希表
